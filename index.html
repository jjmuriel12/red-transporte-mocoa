async function acceptService(){ if(!checkSupabase()) return; if(!currentDemand){ alert('No hay una solicitud.'); return; } if(!selectedPartner){ alert('Selecciona un socio primero.'); return; } const errorBox = document.getElementById('assignmentError'); const button = document.getElementById('assignBtn'); button.disabled = true; button.textContent = '⏳ Verificando disponibilidad...'; errorBox.innerHTML = ''; try{ /* IMPORTANTE: Primero intentamos ocupar al socio SOLO si todavía está disponible. */ const { data: socioActualizado, error: socioError } = await supabaseClient .from('socios') .update({ estado: 'ocupado' }) .eq('id', selectedPartner.id) .eq('estado', 'disponible') .select() .maybeSingle(); if(socioError){ console.error(socioError); errorBox.innerHTML = `
❌ No se pudo verificar el socio. ${escapeHtml( formatSupabaseError(socioError) )}
`; return; } /* Si no regresó ningún registro significa que OTRO USUARIO YA OCUPÓ A ESE SOCIO. */ if(!socioActualizado){ errorBox.innerHTML = `
🔴 ${escapeHtml( selectedPartner.nombre || 'Este socio' )} ya no está disponible.

Otro usuario probablemente acaba de asignarlo a un servicio.
`; selectedPartner = null; await loadPartners(); show('matching'); return; } /* El socio quedó ocupado correctamente. Ahora asignamos la solicitud. */ const { data: solicitudActualizada, error: solicitudError } = await supabaseClient .from('solicitudes') .update({ estado: 'asignada', socio_id: selectedPartner.id }) .eq('id', currentDemand.id) .eq('estado', 'pendiente') .select() .maybeSingle(); if(solicitudError){ /* Si falla la solicitud, devolvemos al socio a disponible para no dejarlo bloqueado. */ await supabaseClient .from('socios') .update({ estado: 'disponible' }) .eq('id', selectedPartner.id); errorBox.innerHTML = `
❌ No se pudo asignar la solicitud. ${escapeHtml( formatSupabaseError(solicitudError) )}
`; return; } /* Si no se actualizó ninguna solicitud, probablemente otro usuario ya la tomó. */ if(!solicitudActualizada){ await supabaseClient .from('socios') .update({ estado: 'disponible' }) .eq('id', selectedPartner.id) .eq('estado', 'ocupado'); errorBox.innerHTML = `
⚠️ Esta solicitud ya fue asignada por otro usuario.
`; selectedPartner = null; await loadPartners(); show('matching'); return; } /* Guardamos la solicitud actualizada. */ currentDemand = solicitudActualizada; /* Mostramos el servicio activo. */ document.getElementById( 'activePartner' ).textContent = selectedPartner.nombre || 'Socio'; document.getElementById( 'deliveredPartner' ).textContent = selectedPartner.nombre || 'Socio'; document.getElementById( 'activeServiceText' ).innerHTML = ` 🚚 ${escapeHtml( selectedPartner.vehiculo || '-' )}

📍 ${escapeHtml( currentDemand.origen || '' )} → ${escapeHtml( currentDemand.destino || '' )}

📦 ${escapeHtml( String( currentDemand.mercancia || '' ) )}
⚖️ ${escapeHtml( String( currentDemand.cantidas ?? '' ) )} ${escapeHtml( currentDemand.unidad || '' )} `; /* Actualizamos inmediatamente la información local y mostramos el servicio. */ await loadPartners(); show('active'); }catch(error){ console.error(error); errorBox.innerHTML = `
❌ Error al asignar el servicio. ${escapeHtml( error.message || String(error) )}
`; }finally{ button.disabled = false; button.textContent = '✅ Asignar servicio'; } }
