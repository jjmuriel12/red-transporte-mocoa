async function openAdmin(){ if(!checkSupabase()) return; try{ const { data, error } = await supabaseClient.auth.getSession(); if(error){ console.error(error); show('adminLogin'); return; } const session = data.session; /* Si no hay sesión: mostrar login. */ if(!session){ show('adminLogin'); return; } /* Verificar que el usuario sea el administrador autorizado. */ const email = String( session.user.email || '' ).toLowerCase().trim(); if(email !== ADMIN_EMAIL){ await supabaseClient.auth.signOut(); document.getElementById( 'adminLoginError' ).innerHTML = `
❌ Este usuario no tiene permiso para entrar a la Central.
`; show('adminLogin'); return; } /* Usuario autorizado. */ show('admin'); loadPartners(); }catch(error){ console.error(error); show('adminLogin'); } }
